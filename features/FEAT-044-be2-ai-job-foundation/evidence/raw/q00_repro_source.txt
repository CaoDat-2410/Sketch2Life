out: dict[str, dict[str, object]] = {}


def rec(rid: str, defect: bool, **observed: object) -> None:
    out[rid] = {"defect_present": defect, "observed": observed}


def guard(rid: str, fn) -> None:  # type: ignore[no-untyped-def]
    try:
        fn()
    except Exception as exc:  # noqa: BLE001
        rec(rid, True, exception=f"{type(exc).__name__}: {exc}")


def r01() -> None:
    h = build_fixture_harness()
    j = h.service.create_job(h.create()).job_id
    m = h.start_next()
    foreign = h.artifacts.put(OTHER_ORG, b"foreign-org")
    other_job = h.artifacts.put(ORG, b"same-org-other-job")
    claim = replace(h.success_claim(m), input_hash="0" * 64, outputs=(foreign, other_job))
    r = h.service.accept_completion(h.capability(j), claim)
    h2 = build_fixture_harness()
    j2 = h2.service.create_job(h2.create()).job_id
    m2 = h2.start_next()
    foreign2 = h2.artifacts.put(OTHER_ORG, b"foreign-org")
    r2 = h2.service.accept_completion(h2.capability(j2),
                                      replace(h2.success_claim(m2), outputs=(foreign2,)))
    q = dict(h.artifacts.quarantined)
    if foreign2.artifact_ref in h2.artifacts.quarantined:
        q["h2:" + foreign2.artifact_ref] = h2.artifacts.quarantined[foreign2.artifact_ref]
    rec("R01", foreign.artifact_ref in q or other_job.artifact_ref in q,
        reasons=[r.reason_code, r2.reason_code], quarantined=sorted(q))

def r03() -> None:
    h = build_fixture_harness()
    j = h.service.create_job(h.create()).job_id
    m = h.next_message()
    cap = h.capability(j)
    claim = h.success_claim(m)
    first = h.service.accept_completion(cap, claim)
    h.service.start_attempt(cap, j, m.attempt_id)
    second = h.service.accept_completion(cap, claim)
    h2 = build_fixture_harness()
    j2 = h2.service.create_job(h2.create()).job_id
    m2 = h2.start_next()
    good = h2.success_claim(m2)
    bad = h2.service.accept_completion(h2.capability(j2), replace(good, input_hash="0" * 64))
    try:
        corrected = str(h2.service.accept_completion(h2.capability(j2), good).disposition)
    except AiJobError as exc:
        corrected = exc.code
    rec("R03", second.disposition.value != "accepted" or corrected != "accepted",
        before_start=first.reason_code, after_start=str(second.disposition),
        mismatch=bad.reason_code, corrected=corrected)

        else resp.text[:60])


def r05() -> None:
    h = build_fixture_harness()
    h.copy_service.register_copy(ProviderCopy("c1", ORG, "s1", CopyKind.OUTPUT, "fixture-deletes"))
    original = h.copy_adapter.request_purge

    def slow(copy):  # type: ignore[no-untyped-def]
        time.sleep(0.3)
        return original(copy)

    h.copy_adapter.request_purge = slow  # type: ignore[method-assign]
    results: list[str | None] = []

    def run() -> None:
        results.append(h.copy_service.purge_scope(ORG, "s1", "same")[0].receipt_id)

    threads = [threading.Thread(target=run) for _ in range(2)]
    for t in threads:
        t.start()
    for t in threads:
        t.join()
    copy = h.copy_registry.get("c1")
    stored = h.copy_registry.get_purge_receipt(f"{ORG}:s1:same:c1")
    calls = len(h.copy_adapter.purge_requests)
    consistent = len(set(results)) == 1 and stored is not None and copy is not None and (
        copy.receipt_id == stored.receipt_id == results[0])
    rec("R05", calls != 1 or not consistent, adapter_calls=calls, returned=sorted(map(str, results)),
        stored=stored.receipt_id if stored else None, copy=copy.receipt_id if copy else None)


def r06() -> None:
    h = build_fixture_harness()

    first = h.copy_service.purge_scope(ORG, "s1", "p1")[0]
    h.copy_registry.save(replace(h.copy_registry.get("c1"), provider_profile="fixture-deletes"))
    second = h.copy_service.purge_scope(ORG, "s1", "p2")[0]
    try:
        replay = h.copy_service.purge_scope(ORG, "s1", "p1")[0]
        replay_obs = {"status": replay.status, "receipt": replay.receipt_id}
        err = None
    except Exception as exc:  # noqa: BLE001
        replay_obs, err = None, f"{type(exc).__name__}: {exc}"
    current = h.copy_registry.get("c1")
    rec("R06", err is not None or current is None or current.status.value != "deleted"
        or len(h.copy_adapter.purge_requests) != 2,
        first=first.status, second=second.status, replay=replay_obs, error=err,
        current=current.status.value if current else None,
        adapter_calls=len(h.copy_adapter.purge_requests))
