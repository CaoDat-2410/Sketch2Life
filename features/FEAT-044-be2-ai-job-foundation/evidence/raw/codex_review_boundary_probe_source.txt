class ConcurrentAdapter:
    def __init__(self):
        self.barrier=Barrier(2,timeout=5)
        self.lock=Lock()
        self.calls=0
    def request_purge(self,copy):
        with self.lock:
            self.calls+=1
            number=self.calls
        self.barrier.wait()
        return PurgeReceipt('receipt_fx_'+str(number),copy.copy_id,ReceiptOutcome.DELETED,True,EvidenceKind.FIXTURE)
    def request_cancel(self,copy):
        pass
